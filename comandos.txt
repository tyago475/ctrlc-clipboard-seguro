================================================================
 CTRL+C — Como correr o programa
================================================================

# 1. Ativar o ambiente virtual (sempre que abrires um terminal novo)

run './requirements.sh'
source .venv/bin/activate


# 2. Correr a aplicacao
# ---------------------

# Demonstracao do ataque (nao precisa de clipboard)
python3 T10_Tiago_Bruno_Diogo_Henrique_Goncalo_Rodrigo_attack.py

# CLI — versao simples (chave so em memoria, perde-se ao sair)
python3 T10_Tiago_Bruno_Diogo_Henrique_Goncalo_Rodrigo_app.py --user alice

# CLI — versao com password mestra (persistente)
python3 T10_Tiago_Bruno_Diogo_Henrique_Goncalo_Rodrigo_app.py --user alice --master

# Interface grafica — versao simples
python3 T10_Tiago_Bruno_Diogo_Henrique_Goncalo_Rodrigo_gui.py --user alice

# Interface grafica — versao com password mestra
python3 T10_Tiago_Bruno_Diogo_Henrique_Goncalo_Rodrigo_gui.py --user alice --master


# 3. Comandos dentro da CLI (prompt "ctrl+c>")
# --------------------------------------------

list             lista as entradas
get <i>          decifra e copia a entrada <i> para o clipboard
check <texto>    verifica se <texto> esteve no historico (via hash)
verify           valida a assinatura digital
flush            grava o historico cifrado agora
help             mostra ajuda
quit             termina
